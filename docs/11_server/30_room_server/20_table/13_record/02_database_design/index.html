      概述#

      前面定义了游戏记录的数据结构, 明确了各类数据的职责及其组织关系, 但并未涉及具体的存储实现

      这里通过数据库表实现上述数据结构, 将游戏记录映射为持久化数据, 并通过关联字段建立数据之间的联系

      桌子数据#

      将TableDjRecord、TableXjRecord 和 TablePlayerRecord 分别映射为独立的数据库表, 通过关联字段建立数据之间的关系

      1. TableDjRecord 与 TableXjRecord 中通过game_record_id 关联
        game_recorrd_id 是一场游戏(一大局)的唯一标志
      2. TableDjRecord/TableXjRecord 与 TablePlayerRecord 通过tabel_record_id 关联
        TableDjRecord.table_record_id 就是game_record_id
        TableXjRecord.table_record_id 由game_record_id + current_game_count 的组合生成
      -- 设置game_record_id 为主键

CREATE TABLE IF NOT EXISTS table_dj_record (
    game_record_id varchar(128) PRIMARY KEY,
    kindid int NOT NULL,
    table_id int NOT NULL,
    table_mode int NOT NULL,
    all_count int NOT NULL,
    player_count int NOT NULL,
    record_time timestamp DEFAULT CURRENT_TIMESTAMP
);
      -- 设置game_rocord_id, current_game_count 为联合主键
-- 建立与table_dj_record 的外键约束

CREATE TABLE IF NOT EXISTS table_xj_record (
    game_record_id varchar(128) NOT NULL,
    table_record_id varchar(128) NOT NULL,
    current_game_count int NOT NULL,

    PRIMARY KEY (game_record_id, current_game_count),
    -- 外键约束
    CONSTRAINT fk_table_xj_record_table_dj_record
        FOREIGN KEY (game_record_id)
        REFERENCES table_dj_record(game_record_id)
);
      -- 设置table_record_id + uid为联合主键

CREATE TABLE IF NOT EXISTS table_player_record (
    table_record_id varchar(128) NOT NULL,
    -- 玩家数据
    uid int NOT NULL,
    chair int NOT NULL,
    score int NOT NULL,

    PRIMARY KEY (table_record_id, uid)
);

      玩家数据#

      玩家数据 与 桌子数据的映射, 通过game_record_id 来关联.
      这里同时增加一些方便过滤查询的字段, 比如增加时间字段, 增加游戏字段(kindid)

      -- 设置uid + recordtime 为联合主键
-- 建立与table_dj_record 的外键约束

CREATE TABLE IF NOT EXISTS player_game_record (
    uid bigint NOT NULL,
    game_record_id varchar(32) NOT NULL,
    kindid int NOT NULL,
    record_time timestamp DEFAULT CURRENT_TIMESTAMP,

    PRIMARY KEY (uid, record_time),
    CONSTRAINT fk_player_game_record_table_dj_record
        FOREIGN KEY (game_record_id)
        REFERENCES table_dj_record(game_record_id)
);
